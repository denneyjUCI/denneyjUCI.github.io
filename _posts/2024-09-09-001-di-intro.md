# What is it?

___

Dependency Injection (or DI from here on out), to quote [James Shore](https://www.jamesshore.com/v2/blog/2006/dependency-injection-demystified), is "a $25 term for a $0.05 concept": passing instance variables.

That's it. No fancy container frameworks needed, just passing a variable to an object.

While James explains it extremely well in the article linked above, I wanted to take a stab at it, for my own learning if nothing else.

## Concrete Example

To clarify, let's take a look at some code. This is intentionally kept simple to keep scope creep to a minimum :)

{: .box-note}
**Note:** Many callsites will use `try?` since we don't care about any thrown errors for a toy example. Please handle errors properly in production.

```swift
import Foundation

class API {
    func loadData() async throws -> String {
        try? await Task.sleep(nanoseconds: 1_000_000_000)
        return "some data"
    }
}

class Example {
    private let api: API
    init() {
        api = API()
    }

    func doStuff() async {
        if let data = try? await api.loadData() {
            print(data)
        }
    }
}

let example = Example()
await example.doStuff() // prints "some data"
```

Let's make one change to the code: making the dependency on the API explicit through the constructor

```swift
import Foundation

class API {
    func loadData() async throws -> String {
        try? await Task.sleep(nanoseconds: 1_000_000_000)
        return "some data"
    }
}

class Example {
    private let api: API
    init(api: API) {
        self.api = api
    }

    func doStuff() async {
        if let data = try? await api.loadData() {
            print(data)
        }
    }
}

let api = API()
let example = Example(api: api)
await example.doStuff() // prints "some data"
```

{: .box-note}
**Note:** The behavior of the system is the same, but passing the object allows us to modify that behavior _without_ the base type or clients being aware of the change.

```swift

// Previous classes omitted for brevity

class APITimingWrapper: API {
    private let wrapped: API
    init(wrapped: API) {
        self.wrapped = wrapped
    }

    override func loadData() async throws -> String {
        let now = Date()
        do {
            let result = try await wrapped.loadData()
            print("Got result in \(Date().timeIntervalSince(now))s")
            return result
        } catch {
            print("Got error in \(Date().timeIntervalSince(now))s")
            return ""
        }
    }
}

let api = API()
let wrapper = APITimingWrapper(wrapped: api)
let example = Example(api: wrapper)
await example.doStuff()
// prints "some data", and also prints how long the request took
```

Many base classes have a better way of being represented in Swift: a protocol! After a bit of refactoring, we get to the following full solution.

```swift
import Foundation

protocol API {
    func loadData() async throws -> String
}

class DelayedAPI: API {
    func loadData() async throws -> String {
        try? await Task.sleep(nanoseconds: 1_000_000_000)
        return "some data"
    }
}

class APITimingWrapper: API {
    private let wrapped: API
    init(wrapped: API) {
        self.wrapped = wrapped
    }

    func loadData() async throws -> String {
        let now = Date()
        do {
            let result = try await wrapped.loadData()
            print("Got result in \(Date().timeIntervalSince(now))s")
            return result
        } catch {
            print("Got error in \(Date().timeIntervalSince(now))s")
            return ""
        }
    }
}

class Example {
    private let api: API
    init(api: API) {
        self.api = api
    }

    func doStuff() async {
        if let data = try? await api.loadData() {
            print(data)
        }
    }
}

let api = DelayedAPI()
let wrapped = APITimingWrapper(wrapped: api)
let example = Example(api: wrapped)
await example.doStuff()
```
